- [ ] **Step 1: 写 `plugins/hookify/core/rules.mjs`**

纯逻辑，零 IO —— 这样它能被单独测，也让 `pretooluse.mjs` 只剩协议。

```js
/**
 * hookify 的纯逻辑：解析一份规则文件、对一次工具调用求值。
 *
 * 没有 IO —— 读文件、读 stdin、写 stdout 全在 hooks/pretooluse.mjs 里。
 * 判定词汇只有 deny / ask（设计 §0 决策 4）：写 allow 是**错误**，不是静默降级。
 */

/** 一条规则说不清自己想干什么时抛这个 —— 调用方按 fail-closed 处理（设计 §4）。 */
export class RuleError extends Error {
  constructor(message) {
    super(message);
    this.name = 'RuleError';
  }
}

const OPERATORS = new Set(['matches', 'contains', 'equals', 'starts_with', 'ends_with']);
const DECISIONS = new Set(['deny', 'ask']);
const KNOWN_KEYS = new Set(['name', 'tool', 'field', 'operator', 'pattern', 'decision', 'message']);
const REQUIRED = ['name', 'field', 'operator', 'pattern', 'message'];

/** `key: value` 一行一条。值去掉一层引号，好让 pattern 里写 `:` / `#` 成为可能。 */
function parseFrontmatter(text, source) {
  const map = {};
  for (const line of text.split(/\r?\n/)) {
    const trimmed = line.trim();
    if (trimmed === '' || trimmed.startsWith('#')) continue;
    const at = trimmed.indexOf(':');
    if (at === -1) {
      throw new RuleError(`${source}: frontmatter 里这一行不是 key: value —— ${trimmed}`);
    }
    const key = trimmed.slice(0, at).trim();
    let value = trimmed.slice(at + 1).trim();
    if (value.length >= 2) {
      const first = value[0];
      if ((first === '"' || first === "'") && value.endsWith(first)) value = value.slice(1, -1);
    }
    map[key] = value;
  }
  return map;
}

/**
 * 解析一份规则文件。返回 `{ rule, warnings }`。
 *
 * 错误（说不清想干什么）抛 RuleError；**未知字段只告警**（设计 §4 的那条例外：
 * 否则给 frontmatter 加一个新字段会让所有规则全变成 deny）。
 */
export function parseRule(text, source) {
  const warnings = [];
  const lines = text.split(/\r?\n/);
  if (lines[0]?.trim() !== '---') {
    throw new RuleError(`${source}: 规则文件必须以 --- 开头的一段 frontmatter 开始`);
  }
  let end = -1;
  for (let i = 1; i < lines.length; i += 1) {
    if (lines[i].trim() === '---') {
      end = i;
      break;
    }
  }
  if (end === -1) throw new RuleError(`${source}: frontmatter 没有闭合的 ---`);

  const fm = parseFrontmatter(lines.slice(1, end).join('\n'), source);
  const body = lines.slice(end + 1).join('\n').trim();

  for (const key of Object.keys(fm)) {
    if (!KNOWN_KEYS.has(key)) warnings.push(`${source}: 不认识的字段 "${key}"，已忽略`);
  }
  for (const key of REQUIRED) {
    if (fm[key] === undefined || fm[key] === '') {
      throw new RuleError(`${source}: 缺少必填字段 "${key}"`);
    }
  }
  if (!OPERATORS.has(fm.operator)) {
    throw new RuleError(
      `${source}: 不认识的 operator "${fm.operator}"（支持 ${[...OPERATORS].join(' / ')}）`,
    );
  }
  const decision = fm.decision ?? 'deny';
  if (!DECISIONS.has(decision)) {
    throw new RuleError(
      `${source}: decision 只能是 deny / ask，给的是 "${decision}"。放行是表达不出来的 —— ` +
        `与第三方插件只能拧紧不能拧松是同一条。`,
    );
  }
  if (fm.operator === 'matches') {
    try {
      new RegExp(fm.pattern);
    } catch (error) {
      throw new RuleError(`${source}: pattern 不是合法的正则：${String(error)}`);
    }
  }

  return {
    rule: {
      name: fm.name,
      tool: fm.tool,
      field: fm.field,
      operator: fm.operator,
      pattern: fm.pattern,
      decision,
      message: fm.message,
      body,
      source,
    },
    warnings,
  };
}

/** matcher：精确名 / `A|B` / `*` / 省略 = 全部。与 hooks.json 的 matcher 同一套语义。 */
function toolMatches(matcher, tool) {
  if (matcher === undefined || matcher === '' || matcher === '*') return true;
  return matcher
    .split('|')
    .map((name) => name.trim())
    .some((name) => name === tool);
}

function test(operator, haystack, pattern) {
  switch (operator) {
    case 'matches':
      return new RegExp(pattern).test(haystack);
    case 'contains':
      return haystack.includes(pattern);
    case 'equals':
      return haystack === pattern;
    case 'starts_with':
      return haystack.startsWith(pattern);
    case 'ends_with':
      return haystack.endsWith(pattern);
    default:
      // parseRule 已经挡掉；真走到这儿说明上面漏了一种算子，宁可当作不匹配也别抛。
      return false;
  }
}

/**
 * 对一次工具调用求值。返回 `{ decision, reason }`，或 `undefined`（放行）。
 *
 * 命中里只要有 deny 就是 deny，否则只要有 ask 就是 ask（设计 §3.4）。
 * `field` **只取字符串**：键不存在、或是数组 / 对象 → 这条规则不匹配，不是错误。
 */
export function evaluate(rules, toolName, input) {
  const hits = [];
  for (const rule of rules) {
    if (!toolMatches(rule.tool, toolName)) continue;
    const value =
      input !== null && typeof input === 'object' ? input[rule.field] : undefined;
    if (typeof value !== 'string') continue;
    if (!test(rule.operator, value, rule.pattern)) continue;
    hits.push(rule);
  }
  if (hits.length === 0) return undefined;

  const decision = hits.some((rule) => rule.decision === 'deny') ? 'deny' : 'ask';
  const decisive = hits.filter((rule) => rule.decision === decision);
  const lines = decisive.map((rule) => `[${rule.name}] ${rule.message}`);
  for (const rule of decisive) {
    if (rule.body !== '') lines.push(`  ${rule.body}`);
  }
  return { decision, reason: lines.join('\n') };
}
```

- [ ] **Step 2: 写 `plugins/hookify/hooks/pretooluse.mjs`**

```js
#!/usr/bin/env node
/**
 * hookify 的 PreToolUse 钩子：stdin JSON → 读规则 → 求值 → stdout JSON / 退出码。
 *
 * 出错一律**退出非零**：框架把非零翻成 deny（src/cli/hook-run.ts:193-195），
 * 所以这是 fail-closed 的方向 —— 与参考实现 hookify 的 `finally: exit(0)` 正好相反（设计 §4）。
 */
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { RuleError, evaluate, parseRule } from '../core/rules.mjs';

const PREFIX = 'hookify.';
const SUFFIX = '.local.md';

async function readStdin() {
  const chunks = [];
  for await (const chunk of process.stdin) chunks.push(chunk);
  return Buffer.concat(chunks).toString('utf8');
}

function die(message) {
  process.stderr.write(`hookify: ${message}\n`);
  process.exit(1);
}

let input;
try {
  input = JSON.parse(await readStdin());
} catch (error) {
  die(`stdin 不是合法 JSON：${String(error)}`);
}

// 规则目录从 stdin 里的 cwd 拼，不从 process.cwd() 拼 ——
// 钩子进程的 cwd 不保证是会话根（设计 §2）。
const cwd = typeof input?.cwd === 'string' ? input.cwd : process.cwd();
const dir = join(cwd, '.deepcode');

let names = [];
try {
  names = (await readdir(dir)).filter((n) => n.startsWith(PREFIX) && n.endsWith(SUFFIX)).sort();
} catch (error) {
  // 目录不存在 = 没有规则 = 常态，不是错。
  if (error?.code !== 'ENOENT') die(`读不了规则目录 ${dir}：${String(error)}`);
}

const rules = [];
for (const name of names) {
  const path = join(dir, name);
  let text;
  try {
    text = await readFile(path, 'utf8');
  } catch (error) {
    die(`读不了规则文件 ${path}：${String(error)}`);
  }
  try {
    const { rule, warnings } = parseRule(text, path);
    for (const warning of warnings) process.stderr.write(`hookify: ${warning}\n`);
    rules.push(rule);
  } catch (error) {
    if (error instanceof RuleError) die(error.message);
    throw error;
  }
}

const verdict = evaluate(rules, String(input?.tool_name ?? ''), input?.tool_input);
if (verdict === undefined) {
  process.exit(0);
}

// ⚠️ 理由是 `permissionDecisionReason`，**不是**参考实现的 `systemMessage` ——
// 后者我们静默忽略（src/cli/hook-run.ts:219）。抄参考实现会得到"拦下了但一句话不说"。
process.stdout.write(
  `${JSON.stringify({
    hookSpecificOutput: {
      hookEventName: 'PreToolUse',
      permissionDecision: verdict.decision,
      permissionDecisionReason: verdict.reason,
    },
  })}\n`,
);
process.exit(0);
```

```ts
import { spawn } from 'node:child_process';
import { mkdir, mkdtemp, rm, writeFile } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join, resolve } from 'node:path';
import { afterEach, describe, expect, it } from 'vitest';

const HOOK = resolve('plugins/hookify/hooks/pretooluse.mjs');

const made: string[] = [];
afterEach(async () => {
  while (made.length > 0) await rm(made.pop()!, { recursive: true, force: true });
});

/** 一个临时工程：`.deepcode/` 下按需写规则文件。返回它的根。 */
async function world(rules: Record<string, string> = {}): Promise<string> {
  const root = await mkdtemp(join(tmpdir(), 'deepcode-hookify-'));
  made.push(root);
  if (Object.keys(rules).length > 0) {
    await mkdir(join(root, '.deepcode'), { recursive: true });
    for (const [name, text] of Object.entries(rules)) {
      await writeFile(join(root, '.deepcode', name), text);
    }
  }
  return root;
}

/** 跑一次钩子。`toolInput` 进 tool_input，其余进 BaseHookInput。 */
async function run(
  root: string,
  tool: string,
  toolInput: unknown,
): Promise<{ code: number; stdout: string; stderr: string }> {
  const input = {
    session_id: 's1',
    transcript_path: join(root, 't.jsonl'),
    cwd: root,
    hook_event_name: 'PreToolUse',
    tool_name: tool,
    tool_input: toolInput,
    tool_use_id: 'u1',
  };
  return await new Promise((resolvePromise, reject) => {
    const child = spawn(process.execPath, [HOOK], { stdio: ['pipe', 'pipe', 'pipe'] });
    let stdout = '';
    let stderr = '';
    child.stdout.on('data', (chunk) => (stdout += String(chunk)));
    child.stderr.on('data', (chunk) => (stderr += String(chunk)));
    child.on('error', reject);
    child.on('close', (code) => resolvePromise({ code: code ?? -1, stdout, stderr }));
    child.stdin.end(JSON.stringify(input));
  });
}

/** 一条规则文件的正文。 */
function ruleFile(over: Partial<Record<string, string>> = {}): string {
  const fm = {
    name: 'no-aws-keys',
    tool: 'Write|Edit',
    field: 'content',
    operator: 'matches',
    pattern: 'AKIA[0-9A-Z]{16}',
    decision: 'deny',
    message: '文件里出现了形似 AWS access key 的东西',
    ...over,
  };
  return `---\n${Object.entries(fm)
    .filter(([, v]) => v !== undefined)
    .map(([k, v]) => `${k}: ${v}`)
    .join('\n')}\n---\n`;
}

/** 从 stdout 里取判定（没有判定时 undefined）。 */
function decisionOf(stdout: string): string | undefined {
  const parsed = JSON.parse(stdout) as {
    hookSpecificOutput?: { permissionDecision?: string; permissionDecisionReason?: string };
  };
  return parsed.hookSpecificOutput?.permissionDecision;
}

function reasonOf(stdout: string): string {
  const parsed = JSON.parse(stdout) as {
    hookSpecificOutput?: { permissionDecisionReason?: string };
  };
  return parsed.hookSpecificOutput?.permissionDecisionReason ?? '';
}

describe('hookify：没有规则时什么都不做', () => {
  it('没有 .deepcode 目录 → 退出 0、stdout 空', async () => {
    const root = await world();
    const out = await run(root, 'Write', { file_path: 'a.txt', content: 'x' });
    expect(out.code).toBe(0);
    expect(out.stdout.trim()).toBe('');
  });

  it('有目录但没有 hookify.*.local.md → 同样退出 0', async () => {
    const root = await world({ 'settings.json': '{}' });
    const out = await run(root, 'Write', { file_path: 'a.txt', content: 'x' });
    expect(out.code).toBe(0);
    expect(out.stdout.trim()).toBe('');
  });
});

describe('hookify：命中与不命中', () => {
  it('命中 → deny，理由里有规则名与 message', async () => {
    const root = await world({ 'hookify.no-keys.local.md': ruleFile() });
    const out = await run(root, 'Write', {
      file_path: 'a.txt',
      content: 'const k = "AKIAIOSFODNN7EXAMPLE";',
    });
    expect(out.code).toBe(0);
    expect(decisionOf(out.stdout)).toBe('deny');
    expect(reasonOf(out.stdout)).toContain('no-aws-keys');
    expect(reasonOf(out.stdout)).toContain('AWS access key');
  });
```

```ts
export type ConfigEntry = {
  readonly key: string;
  readonly value: string;
  readonly source: 'env' | 'user' | 'project' | 'default';
  readonly writable: boolean;
};

export type DescribeConfigDeps = {
  env: Record<string, string | undefined>;
  home: string;
  root: string;
  readText?: (path: string) => Promise<string | undefined>;
};

/**
 * 生效配置的清单，**带来源**。与 `resolveSettings` 共用同一张 `ENV_NAMES` 与同一条优先级
 * （环境变量 > 项目级 > 用户级 > 默认），但多报一个 source —— 那正是这个事件存在的理由。
 *
 * 一个键同时来自环境与文件时**只报 env**，不再把文件里的值列一遍：
 * 那正是 `resolveSettings` 那条告警要防的「你以为配置生效了」（src/cli/config.ts:56-63）。
 */
export async function describeConfig(deps: DescribeConfigDeps): Promise<ConfigEntry[]> {
  const readText = deps.readText ?? readTextFile;
  const layers: Array<{ path: string; label: 'user' | 'project' }> = [
    { path: join(deps.home, '.deepcode', 'config.json'), label: 'user' },
    { path: join(deps.root, '.deepcode', 'config.json'), label: 'project' },
  ];
  const fromFile: Record<string, { value: unknown; source: 'user' | 'project' }> = {};
  for (const { path, label } of layers) {
    const text = await readText(path);
    if (text === undefined) continue;
    // 坏文件照样抛（parseLayer 的语义）—— describe 不该把坏配置藏起来。
    const parsed = JSON.parse(text) as Record<string, unknown>;
    for (const [key, value] of Object.entries(parsed)) {
      if (key in FIELD_TYPES) fromFile[key] = { value, source: label };
    }
  }

  const entries: ConfigEntry[] = [];
  for (const key of Object.keys(FIELD_TYPES) as Array<keyof CliConfig>) {
    const envName = ENV_NAMES.find((e) => e.field === key)?.names.find(
      (name) => (deps.env[name] ?? '') !== '',
    );
    const file = fromFile[key];
    let value: unknown;
    let source: ConfigEntry['source'];
    if (envName !== undefined) {
      value = deps.env[envName];
      source = 'env';
    } else if (file !== undefined) {
      value = file.value;
      source = file.source;
    } else if (key === 'model') {
      value = DEFAULT_MODEL;
      source = 'default';
    } else {
      value = undefined;
      source = 'default';
    }
    const shown =
      key === 'apiKey'
        ? maskSecret(typeof value === 'string' ? value : undefined)
        : value === undefined
          ? '（没配）'
          : String(value);
    entries.push({ key, value: shown, source, writable: WRITABLE_KEYS.includes(key) });
  }
  return entries;
}

export type SetConfigDeps = {
  home: string;
  key: string;
  value: string;
  readText?: (path: string) => Promise<string | undefined>;
  /** 测试替身。默认真写盘（原子 + 0600）。 */
  writeJson?: (path: string, json: string) => Promise<void>;
};

export type SetConfigResult =
  | { readonly ok: true; readonly path: string }
  | { readonly ok: false; readonly error: string };

/**
 * 改**用户级** `~/.deepcode/config.json` 里的一个键（设计 §5.3）。
 *
 * 三条边界，缺一不可：
 * - 只写用户级：项目级那个文件是仓库的，写它正是「key 跟着仓库走」那条路；
 * - 读-改-写：不能留一个「只写了 model、把 apiKey 抹了」的出口；
 * - 原子 + 0600：一次中断留下的是半个 JSON，而 parseLayer 会把它当坏文件抛。
 */
export async function setConfigValue(deps: SetConfigDeps): Promise<SetConfigResult> {
  for (const prefix of FORBIDDEN_PREFIXES) {
    if (deps.key === prefix || deps.key.startsWith(`${prefix}.`) || deps.key.startsWith(`${prefix}:`)) {
      return {
        ok: false,
        error:
          `"${deps.key}" 属于 ${prefix}.*，那是**设置**不是连接配置 —— ` +
          `改权限 / 沙箱 / 插件名单请直接编辑设置文件。一个能在运行时改权限来源的入口，` +
          `就是第二个自动放行开关。`,
      };
    }
  }
  if (!WRITABLE_KEYS.includes(deps.key as keyof CliConfig)) {
    return {
      ok: false,
      error:
        `"${deps.key}" 不在可写清单里。能写的只有：${WRITABLE_KEYS.join(' / ')}。` +
        `（apiKey 是凭据、baseUrl 决定请求发去哪 —— 这两个要自己开文件改。）`,
    };
  }
  if (deps.key === 'model') {
    try {
      assertModelShape(deps.value);
    } catch (error) {
      return { ok: false, error: String(error instanceof Error ? error.message : error) };
    }
  }

  const path = join(deps.home, '.deepcode', 'config.json');
  const readText = deps.readText ?? readTextFile;
  const existing = await readText(path);

  let record: Record<string, unknown> = {};
  if (existing !== undefined) {
    let parsed: unknown;
    try {
      parsed = JSON.parse(existing);
    } catch (error) {
      return { ok: false, error: `现有的 ${path} 不是合法 JSON，改它之前先修它：${String(error)}` };
    }
    if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
      return { ok: false, error: `现有的 ${path} 顶层不是对象，改它之前先修它。` };
    }
    record = { ...(parsed as Record<string, unknown>) };
  }
  record[deps.key] = deps.value;

  const write = deps.writeJson ?? writeJsonAtomic;
  try {
    await write(path, `${JSON.stringify(record, null, 2)}\n`);
  } catch (error) {
    return { ok: false, error: `写 ${path} 失败：${String(error)}` };
  }
  return { ok: true, path };
}

/** 写临时文件再 rename —— 与 P4c-1 的落盘同一条（那份设计 §5）。 */
async function writeJsonAtomic(path: string, json: string): Promise<void> {
  await mkdir(dirname(path), { recursive: true });
  const tmp = `${path}.tmp`;
  await writeFile(tmp, json, { mode: 0o600 });
  await rename(tmp, path);
}
```

```ts
import type { Plugin } from '../contract/plugin';
import { describeConfig, setConfigValue } from './config';

/**
 * `/config` 命令、`config.describe` 与 `config.set` 三个都住这一个插件里。
 *
 * 住在 `cli/` 不住 `builtins/` 的理由与 `cli/commands.ts:14-19` 一字不差：
 * 它要读盘（`describeConfig` / `setConfigValue`），而 `builtins/` 只许 import `contract/`。
 */
export type ConfigCommandDeps = {
  readonly home: string;
  readonly root: string;
  readonly env: Record<string, string | undefined>;
  /** 未知字段 / 覆盖关系的告警出口。默认写 stderr。 */
  readonly warn?: (line: string) => void;
};

export function createConfigCommand(deps: ConfigCommandDeps): Plugin {
  return {
    name: 'config-command',
    uses: ['command.run', 'config.describe', 'config.set', 'ui'],

    register(on) {
      on('command.describe', { name: 'config' }, () => ({
        description: '看生效的连接配置（改它用 deepcode config set）',
      }));

      on('config.describe', async () => ({
        entries: await describeConfig({
          env: deps.env,
          home: deps.home,
          root: deps.root,
        }),
      }));

      on('config.set', async (_$, e) =>
        await setConfigValue({ home: deps.home, key: e.key, value: e.value }),
      );

      on('command.run', { name: 'config' }, async ($) => {
        const entries = await describeConfig({ env: deps.env, home: deps.home, root: deps.root });
        $.ui.log(renderEntries(entries));
        return { handled: true };
      });
    },
  };
}

/** 一屏清单。可写的那几行带一句提示 —— 用户看得见能改什么，才不用去翻文档。 */
export function renderEntries(entries: readonly ConfigEntry[]): string {
  const width = Math.max(...entries.map((entry) => entry.key.length));
  const lines = entries.map((entry) => {
    const key = entry.key.padEnd(width);
    const mark = entry.writable ? '  （可改）' : '';
    return `  ${key}  ${entry.value}  [${entry.source}]${mark}`;
  });
  return ['连接配置：', ...lines, '改它：deepcode config set model <名字>'].join('\n');
}
```

> 少了会在装载期被撤掉（`src/core/third-party.ts:117-124`）。
