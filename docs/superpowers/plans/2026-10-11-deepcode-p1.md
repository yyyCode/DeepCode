### Task 1: 座位顺序显式化

**Files:**
- Modify: `src/core/registry.ts`
- Test: `test/dispatch.test.ts`

**背景**：现在排序键是 `(tierIndex, seq)`，`seq` 是**全局**注册序号。因为 `createEngine`
一个插件接一个插件地注册，全局序号恰好等于"先按插件、再按注册顺序" —— 但这是碰巧。
一个插件只要在运行期懒注册 hook，它的 hook 就会插到别人后面去。

- [ ] **Step 1: 改 `Registration` 与 `Registry`**

```ts
export type Registration = {
  readonly plugin: string;
  readonly tier: Tier;
  readonly pattern: string;
  readonly matcher?: Record<string, unknown>;
  readonly hook: EventHook;
  /**
   * 插件装载顺序：该插件**第一次**注册时分配，之后不再变。
   * 同 tier 内先按它排，把同一个插件的 hook 聚成一组 —— 见 hooksFor。
   */
  readonly pluginOrder: number;
  /** 同一插件内多次注册按注册顺序嵌套，先注册者在最外层。 */
  readonly seq: number;
};
```

`Registry` 加两个私有成员与一个分配函数：

```ts
export class Registry {
  private readonly regs: Registration[] = [];
  private seq = 0;
  /** plugin → 装载顺序。Map 的 size 天然是"下一个空位"，不需要第二个计数器。 */
  private readonly pluginOrders = new Map<string, number>();

  register(input: {
    plugin: string;
    tier: Tier;
    pattern: string;
    matcher?: Record<string, unknown>;
    hook: EventHook;
  }): Registration {
    const reg: Registration = {
      ...input,
      pluginOrder: this.orderOf(input.plugin),
      seq: this.seq++,
    };
    this.regs.push(reg);
    return reg;
  }

  private orderOf(plugin: string): number {
    const known = this.pluginOrders.get(plugin);
    if (known !== undefined) return known;
    const assigned = this.pluginOrders.size;
    this.pluginOrders.set(plugin, assigned);
    return assigned;
  }
  // all() / matcherValues() 不变
```

`hooksFor` 的排序改成三级键：

```ts
  /**
   * 取某事件的 hook，按 tier 由外到内；**同 tier 内按插件装载顺序分组**，
   * 同插件内按注册先后。三级键的顺序不能换：tier 是权威序，比"谁先注册"更高。
   *
   * 中间那一级不是装饰。少了它，一个插件运行期懒注册的 hook 会插到别的插件后面去，
   * 而"同 plugin 先注册者最外层"这条规则就只在"所有插件一次性注册完"时成立 ——
   * 那是个碰巧，不是个保证。
   */
  hooksFor(event: string, e: Record<string, unknown>): readonly Registration[] {
    return this.regs
      .filter((r) => matchesPattern(r.pattern, event) && matchesMatcher(r.matcher, e))
      .sort(
        (a, b) =>
          tierIndex(a.tier) - tierIndex(b.tier) ||
          a.pluginOrder - b.pluginOrder ||
          a.seq - b.seq,
      );
  }
```

- [ ] **Step 1: 抽出 `src/providers/sse.ts`**

把 `anthropic.ts` 底部的 `sseFrames` 整个搬过来，**只改一处**：帧不带 `event:` 行时也产出（`event` 为空串）。

```ts
/**
 * 逐帧读 SSE。
 *
 * 两家的形状不一样：Anthropic 每帧带 `event:` 行，OpenAI 只有 `data:` 行、以 `data: [DONE]` 收尾。
 * 所以这里只管切帧，`event` 允许是空串 —— 认不认空事件由各自的 provider 决定
 * （Anthropic 侧只认有名字的帧，与它原来的行为一致：那些帧本来就被丢掉）。
 */
export async function* sseFrames(
  body: ReadableStream<Uint8Array>,
): AsyncGenerator<{ event: string; data: string }> {
  const decoder = new TextDecoder();
  let buffer = '';

  for await (const chunk of body as unknown as AsyncIterable<Uint8Array>) {
    buffer += decoder.decode(chunk, { stream: true });

    let boundary = buffer.indexOf('\n\n');
    while (boundary !== -1) {
      const raw = buffer.slice(0, boundary);
      buffer = buffer.slice(boundary + 2);

      let event = '';
      const dataLines: string[] = [];
      for (const line of raw.split('\n')) {
        if (line.startsWith('event:')) event = line.slice(6).trim();
        else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim());
      }
      // 只看有没有 data：event 可以缺，data 缺了这帧就没有意义。
      if (dataLines.length > 0) yield { event, data: dataLines.join('\n') };

      boundary = buffer.indexOf('\n\n');
    }
  }
}
```

`anthropic.ts` 删掉自己的那份，改成 `import { sseFrames } from './sse';`。

```ts
/** 响应体里出现这些字样，就基本可以断定是模型 id 写错了。 */
const MODEL_HINT_RE = /model[_ ]?not[_ ]?found|invalid[_ ]?model|does not exist|no such model/i;

/**
 * 非 2xx 的统一措辞。
 *
 * 模型 id 写错在两家都表现为"一个看不懂的 404/400"，而启动期**查不了存在性**
 * （兼容端点的模型清单各不相同，硬编一张表反而会拦住合法模型）。所以这里把它翻译成人话，
 * 而不是假装能在启动期拦住（设计 §6）。
 */
export async function describeHttpFailure(
  label: string,
  response: Response,
  model: string,
): Promise<string> {
  const body = await response.text();
  const hint =
    response.status === 404 || MODEL_HINT_RE.test(body)
```

```ts
import type {
  ContentBlock,
  Message,
  ModelProvider,
  StreamEvent,
  ToolSpec,
} from '../contract/model-provider';
import { describeHttpFailure } from './http';
import { sseFrames } from './sse';

export type OpenAIConfig = {
  apiKey: string;
  model: string;
  /** 要自带版本段（`https://api.deepseek.com/v1`）：不做"猜着补 /v1"的魔法，猜错就是 404。 */
  baseUrl?: string;
  maxTokens?: number;
};

const DEFAULT_BASE_URL = 'https://api.openai.com/v1';

/** OpenAI 的 finish_reason → 中立 stopReason。没列的照原样透传。 */
const STOP_REASONS: Record<string, string> = {
  stop: 'end_turn',
  length: 'max_tokens',
  tool_calls: 'tool_use',
  function_call: 'tool_use',
};

export function createOpenAIProvider(config: OpenAIConfig): ModelProvider {
  const baseUrl = (config.baseUrl ?? DEFAULT_BASE_URL).replace(/\/$/, '');

  return {
    name: 'openai',
    async *stream({ system, messages, tools, signal }) {
      const response = await fetch(`${baseUrl}/chat/completions`, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          authorization: `Bearer ${config.apiKey}`,
        },
        body: JSON.stringify({
          model: config.model,
          max_tokens: config.maxTokens ?? 8192,
          messages: toOpenAIMessages(system, messages),
          ...(tools.length > 0 ? { tools: tools.map(toOpenAITool) } : {}),
          stream: true,
        }),
        signal,
      });

      if (!response.ok) {
        throw new Error(await describeHttpFailure('openai', response, config.model));
      }
      if (!response.body) throw new Error('openai: 响应没有 body');

      // tool_calls 是**分片**来的：index 相同的就是同一个调用，arguments 要一段段接起来。
      const pending = new Map<number, { id: string; name: string; args: string }>();
      let stopReason = 'end_turn';

      for await (const frame of sseFrames(response.body)) {
        const data = frame.data.trim();
        // OpenAI 用这一行收尾。但不能只靠它 —— 有的兼容端点根本不发，循环自然结束也要收场。
        if (data === '[DONE]') break;

        let parsed: any;
        try {
          parsed = JSON.parse(data);
        } catch {
          continue; // 切帧切坏了的残片：跳过，不值得把整个 turn 打死
        }

        // 错误帧必须抛。默不作声地略过，一次 overloaded 就会变成一个"成功的空 turn"：
        // 没有文本、没有工具调用、stopReason 还是 end_turn，用户的话凭空消失。
        // 与 anthropic.ts 里那段注释是同一条教训。
        if (parsed?.error) {
          const type = parsed.error.type ?? parsed.error.code ?? 'error';
          const detail = parsed.error.message;
          throw new Error(`openai 流中错误帧: ${type}${detail ? ` — ${detail}` : ''}`);
        }

        const choice = parsed?.choices?.[0];
        if (!choice) continue;

        const delta = choice.delta ?? {};
        if (typeof delta.content === 'string' && delta.content !== '') {
          yield { type: 'text', text: delta.content } satisfies StreamEvent;
        }

        for (const call of delta.tool_calls ?? []) {
          const index = typeof call.index === 'number' ? call.index : 0;
          const entry = pending.get(index) ?? { id: '', name: '', args: '' };
          if (typeof call.id === 'string') entry.id = call.id;
          if (typeof call.function?.name === 'string') entry.name = call.function.name;
          if (typeof call.function?.arguments === 'string') entry.args += call.function.arguments;
          pending.set(index, entry);
        }

        if (typeof choice.finish_reason === 'string' && choice.finish_reason !== '') {
          stopReason = STOP_REASONS[choice.finish_reason] ?? choice.finish_reason;
        }
      }

      // 工具调用统一在流末尾产出（与 anthropic 侧在 content_block_stop 产出的时机对齐），
      // 按 index 排序：它是模型给的调用顺序。
      for (const [index, entry] of [...pending.entries()].sort((a, b) => a[0] - b[0])) {
        yield {
          type: 'tool_use',
          // 极少数端点不给 id。给一个稳定的替身，总好过让 loop 拿 undefined 去配对结果。
          id: entry.id === '' ? `call_${index}` : entry.id,
          name: entry.name,
          input: parseArguments(entry.args),
        } satisfies StreamEvent;
      }

      yield { type: 'done', stopReason } satisfies StreamEvent;
    },
  };
}

/**
 * arguments 是分片拼出来的 JSON 字符串。解析失败退化成 `{}` ——
 * 工具会因为缺少参数报错，那条 ERROR 会回到模型手里，比整个 turn 崩掉有用。
 */
function parseArguments(raw: string): unknown {
  if (raw.trim() === '') return {};
  try {
    return JSON.parse(raw);
  } catch {
    return {};
  }
}

/** 中立表示 → OpenAI 协议。注意 tool 结果在这边是**一条一个**，不是合并的。 */
function toOpenAIMessages(system: string, messages: readonly Message[]): unknown[] {
  const out: unknown[] = [{ role: 'system', content: system }];

  for (const message of messages) {
    if (message.role === 'user') {
      out.push({ role: 'user', content: message.content });
      continue;
    }

    if (message.role === 'tool') {
      // OpenAI 的协议是 tool_call_id 一条一个，所以中立层合并的一组在这里摊开。
      for (const result of message.results) {
        out.push({ role: 'tool', tool_call_id: result.toolUseId, content: result.content });
      }
      continue;
    }

    const text = message.content
      .filter((block): block is Extract<ContentBlock, { type: 'text' }> => block.type === 'text')
      .map((block) => block.text)
      .join('');
    const toolCalls = message.content
      .filter((block): block is Extract<ContentBlock, { type: 'tool_use' }> => block.type === 'tool_use')
      .map((block) => ({
        id: block.id,
        type: 'function',
        function: { name: block.name, arguments: JSON.stringify(block.input ?? {}) },
      }));

    out.push({
      role: 'assistant',
      // 只有工具调用、没有文本时必须发 null：部分端点把空字符串当成"空回复"直接报错。
      content: text === '' ? null : text,
      ...(toolCalls.length > 0 ? { tool_calls: toolCalls } : {}),
    });
  }

  return out;
}

function toOpenAITool(tool: ToolSpec): unknown {
  return {
    type: 'function',
    function: { name: tool.name, description: tool.description, parameters: tool.inputSchema },
  };
}
```

```ts
export type Spinner = {
  start(label: string): void;
  stop(): void;
  active(): boolean;
};

const FRAMES = ['⠋', '⠙', '⠹', '⠸', '⠼', '⠴', '⠦', '⠧', '⠇', '⠏'];
const INTERVAL_MS = 90;

/**
 * 单行 spinner，只写 stderr。
 *
 * 非 TTY 下是彻底的 no-op：测试、管道、CI 里它一个字节都不该产出。
 */
export function createSpinner(
  stream: NodeJS.WriteStream = process.stderr,
  enabled: boolean = Boolean(stream.isTTY) && process.env.TERM !== 'dumb',
): Spinner {
  let timer: NodeJS.Timeout | undefined;
  let frame = 0;
  let label = '';

  /** 抹掉这一行。半行小圆圈后面接上模型的第一句话是最典型的 TUI 事故。 */
  const erase = (): void => {
    stream.write('\r\u001b[K');
  };

  return {
    start(next) {
      // 已经在转就什么都不做：重复 start 会把 interval 泄漏掉。
      if (!enabled || timer !== undefined) return;
      label = next;

      const tick = (): void => {
        const glyph = FRAMES[frame++ % FRAMES.length]!;
        stream.write(`\r${glyph} ${label}`);
      };
      tick();
      timer = setInterval(tick, INTERVAL_MS);
      // unref：spinner 不该成为"进程退不掉"的理由。
      timer.unref();

      // 进程死在 spinner 转着的时候，退出前把那一行抹掉。start/stop 是配对的，
      // 所以同时最多挂着一个监听。
      process.once('exit', erase);
    },

    stop() {
      if (timer === undefined) return;
      clearInterval(timer);
      timer = undefined;
      process.off('exit', erase);
      erase();
    },

    active: () => timer !== undefined,
  };
}
```

```ts
import { isAbsolute, relative } from 'node:path';
import type { Palette } from './ansi';

export type DiffLine = { readonly kind: 'context' | 'del' | 'add'; readonly text: string };

export type Hunk = {
  readonly oldStart: number;
  readonly oldCount: number;
  readonly newStart: number;
  readonly newCount: number;
  readonly lines: readonly DiffLine[];
};

/** 中段做 DP 的规模上限：超过就退化成"整段替换"。 */
const MAX_DP_CELLS = 1_000_000;
const CONTEXT = 3;

/**
 * 拆行。`a\nb\n` 切出来是 ['a','b','']，末尾那个空串是行尾换行符的产物、不是一行空行 ——
 * 留着它会让"文件末尾有没有换行"变成一行假改动。
 */
export function splitLines(text: string): string[] {
  if (text === '') return [];
  const lines = text.split('\n');
  if (lines[lines.length - 1] === '') lines.pop();
  return lines;
}

/** 落在 root 里就给相对路径；出不去就别拿一串 ../ 糊弄人，直接给绝对路径。 */
export function displayPath(path: string, root: string): string {
  const rel = relative(root, path);
  if (rel === '') return path;
  return rel.startsWith('..') || isAbsolute(rel) ? path : rel;
}

/**
 * 行级 diff。
 *
 * 先砍公共前后缀：改一个文件通常只动了中间几行，砍完 DP 的规模小一个量级
 * （这是它便宜的原因，不是锦上添花）。中段用 LCS DP；规模超上限就整段替换 ——
 * 结果仍然正确，只是不好看，而不好看远好过卡死。
 */
export function diffLines(before: readonly string[], after: readonly string[]): DiffLine[] {
  let head = 0;
  while (head < before.length && head < after.length && before[head] === after[head]) head++;

  let tail = 0;
  while (
    tail < before.length - head &&
    tail < after.length - head &&
    before[before.length - 1 - tail] === after[after.length - 1 - tail]
  ) {
    tail++;
  }

  const oldMid = before.slice(head, before.length - tail);
  const newMid = after.slice(head, after.length - tail);

  const out: DiffLine[] = before.slice(0, head).map((text) => ({ kind: 'context' as const, text }));

  if (oldMid.length === 0) {
    for (const text of newMid) out.push({ kind: 'add', text });
  } else if (newMid.length === 0) {
    for (const text of oldMid) out.push({ kind: 'del', text });
  } else if (oldMid.length * newMid.length > MAX_DP_CELLS) {
    for (const text of oldMid) out.push({ kind: 'del', text });
    for (const text of newMid) out.push({ kind: 'add', text });
  } else {
    out.push(...lcsDiff(oldMid, newMid));
  }

  for (const text of before.slice(before.length - tail)) out.push({ kind: 'context', text });
  return out;
}

function lcsDiff(oldLines: readonly string[], newLines: readonly string[]): DiffLine[] {
  const n = oldLines.length;
  const m = newLines.length;
  const width = m + 1;

  // Uint32Array 而不是 number[][]：一百万格用嵌套数组会真的吃掉几十兆。
  const table = new Uint32Array((n + 1) * width);
  // noUncheckedIndexedAccess 下 typed array 的索引也带 undefined，收成两个小访问器，
  // 免得等式里到处撒 `!` 和 `??`。
  const at = (i: number, j: number): number => table[i * width + j] ?? 0;
  const put = (i: number, j: number, value: number): void => {
    table[i * width + j] = value;
  };

  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      put(
        i,
        j,
        oldLines[i] === newLines[j]
          ? at(i + 1, j + 1) + 1
          : Math.max(at(i + 1, j), at(i, j + 1)),
      );
    }
  }

  const out: DiffLine[] = [];
  let i = 0;
  let j = 0;
  while (i < n && j < m) {
    if (oldLines[i] === newLines[j]) {
      out.push({ kind: 'context', text: oldLines[i]! });
      i++;
    } else if (at(i + 1, j) >= at(i, j + 1)) {
      out.push({ kind: 'del', text: oldLines[i]! });
      i++;
    } else {
      out.push({ kind: 'add', text: newLines[j]! });
      j++;
    }
  }
  while (i < n) out.push({ kind: 'del', text: oldLines[i++]! });
  while (j < m) out.push({ kind: 'add', text: newLines[j++]! });
  return out;
}

/**
 * 按上下文切 hunk。两处改动的间隔 ≤ 2*context 就并进同一个 —— 与 `git diff` 同口径。
 */
export function toHunks(lines: readonly DiffLine[], context: number = CONTEXT): Hunk[] {
  const changed = lines
    .map((line, index) => (line.kind === 'context' ? -1 : index))
    .filter((index) => index >= 0);
  if (changed.length === 0) return [];

  const hunks: Hunk[] = [];
  let start = Math.max(0, changed[0]! - context);
  let end = Math.min(lines.length, changed[0]! + context + 1);

  for (const index of changed.slice(1)) {
    if (index - context <= end) {
      end = Math.min(lines.length, index + context + 1);
      continue;
    }
    hunks.push(buildHunk(lines, start, end));
    start = Math.max(0, index - context);
    end = Math.min(lines.length, index + context + 1);
  }

  hunks.push(buildHunk(lines, start, end));
  return hunks;
}

function buildHunk(lines: readonly DiffLine[], start: number, end: number): Hunk {
  const slice = lines.slice(start, end);

  let oldCount = 0;
  let newCount = 0;
  for (const line of slice) {
    if (line.kind !== 'add') oldCount++;
    if (line.kind !== 'del') newCount++;
  }

  return {
    oldStart: hunkStart(lines, start, 'add', oldCount),
    oldCount,
    newStart: hunkStart(lines, start, 'del', newCount),
    newCount,
    lines: slice,
  };
}

/**
 * 这段 hunk 的起始行号。
 *
 * 一侧计数为 0 时起始号写 0（`@@ -0,0 +1,3 @@` 表示"从空文件长出来的"），
 * 这是 unified diff 的约定，不是 off-by-one。
 */
function hunkStart(
  lines: readonly DiffLine[],
  index: number,
  skip: 'add' | 'del',
  count: number,
): number {
  let before = 0;
  for (let i = 0; i < index; i++) {
    if (lines[i]!.kind !== skip) before++;
  }
  return count === 0 ? before : before + 1;
}

export type FormatOptions = { readonly isNew?: boolean };

export function formatUnified(
  path: string,
  hunks: readonly Hunk[],
  palette: Palette,
  options: FormatOptions = {},
): string {
  const out = [
    palette.bold(`--- ${options.isNew ? '/dev/null' : `a/${path}`}`),
    palette.bold(`+++ b/${path}`),
  ];

  for (const hunk of hunks) {
    out.push(palette.cyan(`@@ -${hunk.oldStart},${hunk.oldCount} +${hunk.newStart},${hunk.newCount} @@`));
    for (const line of hunk.lines) {
      if (line.kind === 'add') out.push(palette.green(`+${line.text}`));
      else if (line.kind === 'del') out.push(palette.red(`-${line.text}`));
      else out.push(` ${line.text}`);
    }
  }

  return out.join('\n');
}
```

  `/diff` 要看的是"这一整场会话对这个文件的净改动"。
- `snapshot($, path)`：

```ts
      for (const tool of ['Write', 'Edit'] as const) {
        on('tool.call', { tool }, async ($, e, next) => {
          const target = filePathOf(e.input);
          if (target === undefined) return next(e);

          // 必须先读 before 再 next：next 回来的时候文件已经写完了。
          const path = resolve(target);

          const result = await next(e);
          // 只记成功。失败（含"覆盖已有文件前必须先 Read"那道拦截）不留痕迹，

        });
```

```ts
export type ResumeTarget =
  | { readonly kind: 'none' }
  | { readonly kind: 'latest' }
  | { readonly kind: 'id'; readonly id: string };

export type StartupAction =
  | { kind: 'help' }
  | { kind: 'version' }
  | { kind: 'prompt'; readonly text: string; readonly resume: ResumeTarget };

const RESUME = '--resume';

export function parseArgs(argv: readonly string[]): StartupAction {
  // help 优先于 version，两者都优先于 resume 与提示词。
  if (argv.some((arg) => HELP_FLAGS.includes(arg))) return { kind: 'help' };
  if (argv.some((arg) => VERSION_FLAGS.includes(arg))) return { kind: 'version' };

  let resume: ResumeTarget = { kind: 'none' };
  const rest: string[] = [];

  for (const arg of argv) {
    if (arg === RESUME) {
      resume = { kind: 'latest' };
      continue;
    }
    // 只认 `--resume=<id>`。写成"--resume 后面跟一个裸 token"的话，
    // `deepcode --resume "继续修"` 里那句话到底是 id 还是提示词，语法上分不开。
    if (arg.startsWith(`${RESUME}=`)) {
      const id = arg.slice(RESUME.length + 1).trim();
      resume = id === '' ? { kind: 'latest' } : { kind: 'id', id };
      continue;
    }
    // 认出来的开关要从提示词里摘掉，否则 "--resume" 会被当成话发给模型。
    rest.push(arg);
  }

  return { kind: 'prompt', text: rest.join(' ').trim(), resume };
}
```

```ts
async function describeResumeFailure(home: string, cwd: string, id: string | undefined): Promise<string> {
  const all = await listTranscripts(home, cwd);
  const where = `（找的是 ${sessionsDir(home, cwd)}）`;
  if (all.length === 0) {

}
```
